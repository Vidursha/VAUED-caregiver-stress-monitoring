import styles from "../styles/DashboardCharts.module.css";
import {
  Bar,
  BarChart,
  CartesianGrid,
  Legend,
  ResponsiveContainer,
  Tooltip,
  XAxis,
  YAxis,
} from "recharts";
import type { SensorRecord } from "../types";
import { monthFromRecord } from "../utils/heatmap";

type StressGroup = "Low Stress" | "Medium Stress" | "High Stress";

export type ComparisonRow = {
  level: StressGroup;
  EDA: number;
  HR: number;
  TEMP: number;
  MovementMagnitude: number;
};

type Props = {
  allRecords: SensorRecord[];
  month: string;
  months: string[];
  onMonthChange: (month: string) => void;
  onAskAi: (month: string) => void;
};

const COLORS = {
  // Palette from UI reference
  EDA: "#1F6F5F",
  HR: "#2FA084",
  TEMP: "#6FCF97",
  MovementMagnitude: "#EEEEEE",
} as const;

function levelName(label: number): StressGroup {
  if (label === 0) return "Low Stress";
  if (label === 1) return "Medium Stress";
  return "High Stress";
}

function formatMonthLabel(ym: string) {
  const [y, m] = (ym || "").split("-");
  const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
  const idx = Math.max(0, Math.min(11, Number(m) - 1));
  if (!y || !m) return ym;
  return `${months[idx]} ${y}`;
}

function mean(values: number[]) {
  if (!values.length) return 0;
  return values.reduce((a, b) => a + b, 0) / values.length;
}

function normalizeRows(rows: ComparisonRow[]) {
  const keys: (keyof Omit<ComparisonRow, "level">)[] = ["EDA", "HR", "TEMP", "MovementMagnitude"];
  const maxByKey: Record<string, number> = {};
  for (const k of keys) {
    maxByKey[k] = Math.max(...rows.map((r) => Number(r[k] ?? 0)), 1);
  }
  return rows.map((r) => ({
    level: r.level,
    EDA: (r.EDA / maxByKey.EDA) * 100,
    HR: (r.HR / maxByKey.HR) * 100,
    TEMP: (r.TEMP / maxByKey.TEMP) * 100,
    MovementMagnitude: (r.MovementMagnitude / maxByKey.MovementMagnitude) * 100,
  }));
}

function buildExplanation(indexed: ComparisonRow[]) {
  const keys: (keyof Omit<ComparisonRow, "level">)[] = ["EDA", "HR", "TEMP", "MovementMagnitude"];
  const ranges = keys.map((k) => {
    const vals = indexed.map((r) => r[k]);
    return { k, range: Math.max(...vals) - Math.min(...vals) };
  });
  ranges.sort((a, b) => b.range - a.range);
  const mostChanged = ranges[0]?.k ?? "EDA";
  const mostStable = ranges[ranges.length - 1]?.k ?? "TEMP";

  return {
    mostChanged,
    mostStable,
    text:
      `In this view, each signal is shown as an indexed score (0–100) so different units can be compared fairly. ` +
      `The signal that changes the most across stress levels is ${mostChanged}, while ${mostStable} is the most stable. ` +
      `For ward supervisors, large increases across levels may indicate workload strain patterns worth monitoring.`,
  };
}

export function ComparisonChart({ allRecords, month, months, onMonthChange, onAskAi }: Props) {
  const records = allRecords.filter((record) => monthFromRecord(record) === month);
  const { rowsRaw, countsByLevel } = (() => {
    const buckets: Record<StressGroup, SensorRecord[]> = {
      "Low Stress": [],
      "Medium Stress": [],
      "High Stress": [],
    };
    for (const r of records) {
      buckets[levelName(Number(r.label))].push(r);
    }
    const mk = (level: StressGroup, items: SensorRecord[]): ComparisonRow => ({
      level,
      EDA: mean(items.map((x) => Number(x.EDA))),
      HR: mean(items.map((x) => Number(x.HR))),
      TEMP: mean(items.map((x) => Number(x.TEMP))),
      MovementMagnitude: mean(items.map((x) => Number(x.MovementMagnitude))),
    });
    return {
      rowsRaw: [
        mk("Low Stress", buckets["Low Stress"]),
        mk("Medium Stress", buckets["Medium Stress"]),
        mk("High Stress", buckets["High Stress"]),
      ],
      countsByLevel: {
        low: buckets["Low Stress"].length,
        medium: buckets["Medium Stress"].length,
        high: buckets["High Stress"].length,
      },
    };
  })();

  const indexedRows = normalizeRows(rowsRaw);
  const expl = buildExplanation(indexedRows);
  const missingNote =
    countsByLevel.medium === 0
      ? "Note: there are 0 Medium-stress (label 1) readings in this month, so the Medium bars are zero-height."
      : countsByLevel.medium <= 3
        ? `Note: Medium-stress has very few readings (${countsByLevel.medium}), so values may look small or unstable.`
        : "";

  return (
    <section className={styles.chartCard}>
      <div className={styles.chartHeader}>
        <div>
          <h3>Average Physiological Signals Across Stress Levels</h3>
          <p>
            Compare monthly signal shifts across Low, Medium, and High stress groups in one indexed view.
          </p>
        </div>
        <div className={styles.headerActions}>
          <select className={styles.smallSelect} value={month} onChange={(e) => onMonthChange(e.target.value)}>
            {months.map((m) => (
              <option key={m} value={m}>
                {formatMonthLabel(m)}
              </option>
            ))}
          </select>
          <button className={styles.askButton} onClick={() => onAskAi(month)}>
            Ask AI
          </button>
        </div>
      </div>

      <div className={styles.rechartsWrap}>
        <ResponsiveContainer width="100%" height={320}>
          <BarChart data={indexedRows} margin={{ top: 12, right: 20, left: 10, bottom: 10 }} barGap={8} barCategoryGap="20%">
            <CartesianGrid stroke="#e8eef5" strokeDasharray="3 3" />
            <XAxis
              dataKey="level"
              tick={{ fill: "#5b748c", fontSize: 12 }}
              axisLine={{ stroke: "#d7e2ee" }}
              tickLine={{ stroke: "#d7e2ee" }}
            />
            <YAxis
              domain={[0, 110]}
              tickFormatter={(v) => `${v}`}
              tick={{ fill: "#5b748c", fontSize: 12 }}
              axisLine={{ stroke: "#d7e2ee" }}
              tickLine={{ stroke: "#d7e2ee" }}
              label={{
                value: "Indexed score (0–100)",
                angle: -90,
                position: "insideLeft",
                fill: "#6b849b",
                fontSize: 12,
              }}
            />
            <Tooltip
              content={(props: any) => {
                const { active, payload, label } = props;
                if (!active || !payload?.length) return null;
                const row = payload[0].payload as ComparisonRow;
                return (
                  <div className={styles.tooltip}>
                    <div className={styles.tooltipTitle}>{label}</div>
                    {(["EDA", "HR", "TEMP", "MovementMagnitude"] as const).map((k) => (
                      <div key={k} className={styles.tooltipRow}>
                        <span className={styles.dot} style={{ background: COLORS[k] }} />
                        <span>{k}:</span>
                        <strong>{row[k].toFixed(0)}</strong>
                      </div>
                    ))}
                  </div>
                );
              }}
              cursor={{ fill: "rgba(148, 163, 184, 0.12)" }}
            />
            <Legend
              verticalAlign="top"
              align="right"
              iconType="circle"
              wrapperStyle={{ paddingBottom: 6 }}
              formatter={(value) => <span style={{ color: "#4c657e", fontSize: 12 }}>{value}</span>}
            />
            <Bar dataKey="EDA" fill={COLORS.EDA} name="EDA" radius={[10, 10, 0, 0]} barSize={16} />
            <Bar dataKey="HR" fill={COLORS.HR} name="HR" radius={[10, 10, 0, 0]} barSize={16} />
            <Bar dataKey="TEMP" fill={COLORS.TEMP} name="TEMP" radius={[10, 10, 0, 0]} barSize={16} />
            <Bar
              dataKey="MovementMagnitude"
              fill={COLORS.MovementMagnitude}
              name="Movement"
              stroke="#cfd7e3"
              strokeWidth={1}
              radius={[10, 10, 0, 0]}
              barSize={16}
            />
          </BarChart>
        </ResponsiveContainer>
      </div>

      <div className={styles.explainBox}>
        <h4>How to read this (for ward supervisors)</h4>
        <p>
          {records.length
            ? expl.text
            : `No readings are available for ${formatMonthLabel(month)} in this comparison view.`}
        </p>
        {missingNote ? <p style={{ marginTop: 8 }}>{missingNote}</p> : null}
      </div>
    </section>
  );
}

import styles from "../styles/KpiCards.module.css";
import { toFixed } from "../utils/format";

type Props = {
  caregivers: number;
  averageTemperature: number | null;
  stressIndex: number | null;
  averageHeartRate: number | null;
};

export function KpiCards({
  caregivers,
  averageTemperature,
  stressIndex,
  averageHeartRate,
}: Props) {
  const cards = [
    {
      title: "Distinct caregivers",
      value: caregivers,
      note: "IDs in selected month",
      tone: "teal",
    },
    {
      title: "Average temperature",
      value: `${toFixed(averageTemperature)} C`,
      note: "Sensor TEMP mean",
      tone: "sage",
    },
    {
      title: "Stress index",
      value: stressIndex === null ? "-" : `${Math.round(stressIndex)}%`,
      note: "Label-based index",
      tone: "amber",
    },
    {
      title: "Average heart rate",
      value: `${toFixed(averageHeartRate, 1)} bpm`,
      note: "Sensor HR mean",
      tone: "rose",
    },
  ];

  return (
    <section className={styles.grid}>
      {cards.map((card) => (
        <article key={card.title} className={`${styles.card} ${styles[card.tone]}`}>
          <p className={styles.label}>{card.title}</p>
          <h3 className={styles.value}>{card.value}</h3>
          <span className={styles.note}>{card.note}</span>
        </article>
      ))}
    </section>
  );
}
